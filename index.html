05Coursework · FPGA课程设计
独立完成 · Verilog · DE2-115 · ~2800 LE

VGA 贪吃蛇

DE2-115 开发板，Verilog 实现。640×480@60Hz VGA 显示，分四个模块（游戏主逻辑 / VGA 时序驱动 / 屏幕分数显示 / 七段译码），蛇身最长 128 节、吃豆自动加速、伪随机生成与碰撞检测。

VerilogFPGA
// VGA 驱动 · 640×480@60Hz 时序（真实代码）
module vga_driver(input clk_25m, input rst_n,
                  output vga_hs, output vga_vs,
                  output [9:0] pos_x, pos_y);
    parameter H_ACTIVE=640, H_SYNC=96, H_TOTAL=800;
    parameter V_ACTIVE=480, V_SYNC=2,  V_TOTAL=525;
    reg [9:0] h_cnt, v_cnt;
    always @(posedge clk_25m or negedge rst_n)
        if(!rst_n) h_cnt <= 0;
        else if(h_cnt == H_TOTAL-1) h_cnt <= 0;
        else h_cnt <= h_cnt + 1;
    assign vga_hs = !(h_cnt >= H_ACTIVE+H_FRONT
        && h_cnt < H_ACTIVE+H_FRONT+H_SYNC);
    assign pos_x = (h_cnt < H_ACTIVE) ? h_cnt : 10'd0;
endmodule
